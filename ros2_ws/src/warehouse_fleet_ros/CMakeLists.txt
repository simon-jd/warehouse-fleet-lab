cmake_minimum_required(VERSION 3.20)
project(warehouse_fleet_ros)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Assignment.msg" "srv/AllocateTasks.srv")
get_filename_component(FLEET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(fleet_adapter_core STATIC ${FLEET_ROOT}/src/map.cpp ${FLEET_ROOT}/src/task_allocator.cpp ${FLEET_ROOT}/src/reservation_planner.cpp)
target_include_directories(fleet_adapter_core PUBLIC ${FLEET_ROOT}/include)
target_compile_options(fleet_adapter_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(allocator_node src/allocator_node.cpp)
ament_target_dependencies(allocator_node rclcpp)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(allocator_node fleet_adapter_core "${cpp_typesupport_target}")
target_compile_options(allocator_node PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS allocator_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS ${FLEET_ROOT}/scripts/sil/command_gate.py ${FLEET_ROOT}/scripts/sil/coordinator.py DESTINATION lib/${PROJECT_NAME})
install(FILES ${FLEET_ROOT}/scripts/sil/world.py ${FLEET_ROOT}/scripts/sil/protocol.py DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
