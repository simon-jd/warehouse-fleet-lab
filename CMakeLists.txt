cmake_minimum_required(VERSION 3.20)
project(warehouse_fleet_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLEET_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(fleet_core src/map.cpp src/task_allocator.cpp src/reservation_planner.cpp src/simulator.cpp src/event_log.cpp)
target_include_directories(fleet_core PUBLIC include)
target_compile_options(fleet_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FLEET_SANITIZE)
  target_compile_options(fleet_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(fleet_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(fleet src/main.cpp)
target_link_libraries(fleet PRIVATE fleet_core)
target_compile_options(fleet PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(fleet_tests tests/test_core.cpp)
  target_link_libraries(fleet_tests PRIVATE fleet_core)
  add_test(NAME core COMMAND fleet_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_oracle.py $<TARGET_FILE:fleet> ${CMAKE_SOURCE_DIR})
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:fleet> ${CMAKE_SOURCE_DIR})
endif()
